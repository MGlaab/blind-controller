async function connectBLE() { try { bleDevice = await navigator.bluetooth.requestDevice({ filters: [{ name: "ESP32C3_Time_Control" }], optionalServices: [SERVICE_UUID] }); bleDevice.addEventListener('gattserverdisconnected', onDisconnected); const server = await bleDevice.gatt.connect(); const service = await server.getPrimaryService(SERVICE_UUID); charStatus = await service.getCharacteristic(CHAR_STATUS_UUID); charOpen = await service.getCharacteristic(CHAR_SET_OPEN_UUID); charClose = await service.getCharacteristic(CHAR_SET_CLOSE_UUID); // 1. First, attach the listener BEFORE starting notifications charStatus.addEventListener('characteristicvaluechanged', handleStatusUpdate); await charStatus.startNotifications(); // 2. Explicitly READ the stored value directly from ESP32 memory const value = await charStatus.readValue(); parseAndDisplayStatus(value); updateUI(true); } catch (error) { console.error("BLE Connect Error:", error); } } // Function to decode raw BLE bytes to UI function parseAndDisplayStatus(dataView) { const decoder = new TextDecoder('utf-8'); const jsonStr = decoder.decode(dataView); console.log("Received from ESP32 Flash:", jsonStr); try { const data = JSON.parse(jsonStr); currentOpenStr = data.open; currentCloseStr = data.close; // Update screen display elements document.getElementById("disp-open").innerText = currentOpenStr; document.getElementById("disp-close").innerText = currentCloseStr; // Pre-fill input boxes with current values document.getElementById("input-open").value = parse12To24(currentOpenStr); document.getElementById("input-close").value = parse12To24(currentCloseStr); } catch(e) { console.error("Error parsing JSON string:", e); } } function handleStatusUpdate(event) { parseAndDisplayStatus(event.target.value); }
